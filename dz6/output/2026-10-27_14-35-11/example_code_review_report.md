import requests

BASE_URL = "http://localhost:8000"

@pytest.fixture
def base_url():
    return BASE_URL

@pytest.mark.parametrize("payload, expected_status_code, expected_name", [
    ({"name": "Alice", "email": "alice@example.com"}, 201, "Alice"),
    ({"name": "Bob", "email": "bob@example.com"}, 201, "Bob"),
    ({"name": "Charlie", "email": "charlie@example.com"}, 201, "Charlie"),
    ({"name": "Test", "email": "bad-email"}, 400, None),
    ({"email": "test@example.com"}, 400, None)
])
def test_create_user(base_url, payload, expected_status_code, expected_name):
    response = requests.post(f"{base_url}/users", json=payload)
    assert response.status_code == expected_status_code
    if expected_name:
        assert response.json()["name"] == expected_name
    else:
        assert response.status_code >= 400  # Обработка ошибки
